```bash
python3 - <<'PY'
import json, glob, os, statistics
from datetime import datetime
TS = "%Y-%m-%d %H:%M"

def p(v):
    try: return datetime.strptime(v, TS)
    except Exception: return None

def rows_of(s):                      # 규칙 1~4
    a = p(s.get("created_at")); out = []
    if a is None: return out
    for r in s.get("rows", []):
        if r.get("status") != "done": continue          # 규칙 2
        t = p(r.get("timestamp"))
        if t is None: continue                          # 규칙 3
        d = max(0, int((t - a).total_seconds() // 60))   # 규칙 4
        out.append((r.get("stage"), d,
                    "wait" if r.get("owner") == "user" else "work"))  # 규칙 5
        a = max(a, t)                                   # 단조 앵커
    return out

states = []
for f in sorted(glob.glob("tasks/*/state.json")):
    s = json.load(open(f)); s["_tid"] = os.path.basename(os.path.dirname(f)); states.append(s)
done = [s for s in states if s.get("current_status") == "done"]     # 규칙 6

# §2 코호트
by = {}
for s in done: by.setdefault(s["skill"], []).append(s["_tid"])
for k in sorted(by): print(k, len(by[k]), sorted(by[k]))

# §4 워크플로우별
for k in sorted(by):
    tot = []; work = wait = 0
    for s in [x for x in done if x["skill"] == k]:
        rs = rows_of(s); tot.append(sum(d for _, d, _ in rs))
        work += sum(d for _, d, sr in rs if sr == "work")
        wait += sum(d for _, d, sr in rs if sr == "wait")
    print(k, len(tot), statistics.median(tot), work, wait, wait / (work + wait) * 100)

# §3 태스크 단위 (101)
s101 = [s for s in states if s["_tid"].startswith("101")][0]
rs = rows_of(s101)
print("101", sum(d for _, d, _ in rs),
      sum(d for _, d, sr in rs if sr == "work"),
      sum(d for _, d, sr in rs if sr == "wait"),
      len(s101["rows"]), sum(1 for r in s101["rows"] if r.get("gate")))
PY
```
